Basic Bot Example

This example demonstrates a simple Discord bot with common features.

Complete Example

using Nobelium

# Create the bot
bot = Bot(
    token = ENV["DISCORD_TOKEN"],
    intents = Intents.DEFAULT | Intents.MESSAGE_CONTENT
)

# Track bot startup
on!(bot, "READY") do data
    user = data["user"]
    println("✓ Logged in as $(user["username"])#$(user["discriminator"])")
    println("✓ Bot ID: $(user["id"])")
end

# Simple ping command
on!(bot, "MESSAGE_CREATE") do data
    content = get(data, "content", "")
    
    if content == "!ping"
        reply(bot, data; content = "Pong! 🏓")
    end
end

# Echo command
on!(bot, "MESSAGE_CREATE") do data
    content = get(data, "content", "")
    
    if startswith(content, "!echo ")
        text = content[7:end]
        reply(bot, data; content = text)
    end
end

# User info command
on!(bot, "MESSAGE_CREATE") do data
    content = get(data, "content", "")
    
    if content == "!whoami"
        author = data["author"]
        embed = Dict(
            "title" => "User Info",
            "fields" => [
                Dict("name" => "Username", "value" => author["username"], "inline" => true),
                Dict("name" => "ID", "value" => author["id"], "inline" => true),
            ],
            "color" => 0x5865F2
        )
        reply(bot, data; embeds = [embed])
    end
end

# Start the bot
println("Starting bot...")
start!(bot)

Running the Bot

export DISCORD_TOKEN="your-token-here"
julia basic_bot.jl

Adding More Commands

Help Command

const COMMANDS = Dict(
    "!ping" => "Check if the bot is alive",
    "!echo <text>" => "Echo back your message",
    "!whoami" => "Display your user info",
    "!help" => "Show this help message"
)

on!(bot, "MESSAGE_CREATE") do data
    if data["content"] == "!help"
        help_text = join(["**$cmd** - $desc" for (cmd, desc) in COMMANDS], "\n")
        reply(bot, data; content = help_text)
    end
end

Reaction Handling

on!(bot, "MESSAGE_REACTION_ADD") do data
    emoji = data["emoji"]["name"]
    user_id = data["user_id"]
    
    # Don't respond to bot's own reactions
    if user_id == string(bot.user.id)
        return
    end
    
    if emoji == "👍"
        channel_id = data["channel_id"]
        send_message(bot, channel_id; content = "Thanks for the thumbs up!")
    end
end

Message Formatting

using Nobelium: bold, italic, code, code_block, mention_user

on!(bot, "MESSAGE_CREATE") do data
    if data["content"] == "!format"
        msg = """
        $(bold("Bold text"))
        $(italic("Italic text"))
        $(code("inline code"))
        $(code_block("println(\"Hello!\")", language="julia"))
        """
        reply(bot, data; content = msg)
    end
end

Error Handling

on!(bot, "MESSAGE_CREATE") do data
    if startswith(data["content"], "!delete ")
        try
            msg_id = data["content"][9:end]
            delete_message(bot, data["channel_id"], msg_id)
            reply(bot, data; content = "Message deleted!")
        catch e::APIError
            if e.status_code == 403
                reply(bot, data; content = "I don't have permission to delete that message.")
            elseif e.status_code == 404
                reply(bot, data; content = "Message not found.")
            else
                reply(bot, data; content = "Error: $(e.message)")
            end
        end
    end
end

Async Operations

For long-running operations, use async tasks:

on!(bot, "MESSAGE_CREATE") do data
    if data["content"] == "!slow"
        reply(bot, data; content = "Processing...")
        
        @async begin
            sleep(5)  # Simulate long operation
            send_message(bot, data["channel_id"]; content = "Done!")
        end
    end
end
