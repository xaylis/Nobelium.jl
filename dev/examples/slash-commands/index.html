Slash Commands Example

This example demonstrates how to create and handle Discord slash commands.

Registering Commands

using Nobelium

bot = Bot(
    token = ENV["DISCORD_TOKEN"],
    intents = Intents.DEFAULT
)

# Wait for bot to be ready before registering commands
on!(bot, "READY") do data
    # Build a simple command
    ping_cmd = command("ping", "Check if the bot is alive")
    
    # Build a command with options
    greet_cmd = command("greet", "Greet a user") do cmd
        option(cmd, "user", "The user to greet";
            type = ApplicationCommandOptionTypes.USER,
            required = true
        )
        option(cmd, "message", "Custom greeting message";
            type = ApplicationCommandOptionTypes.STRING,
            required = false
        )
    end
    
    # Register commands globally
    register_commands!(bot, [ping_cmd, greet_cmd])
    
    println("Commands registered!")
end

Handling Interactions

on!(bot, "INTERACTION_CREATE") do data
    # Check if it's a slash command
    if data["type"] != InteractionTypes.APPLICATION_COMMAND
        return
    end
    
    cmd_name = data["data"]["name"]
    
    if cmd_name == "ping"
        respond!(bot, data; content = "Pong! 🏓")
        
    elseif cmd_name == "greet"
        options = get(data["data"], "options", [])
        
        # Extract options
        user_id = nothing
        message = "Hello"
        
        for opt in options
            if opt["name"] == "user"
                user_id = opt["value"]
            elseif opt["name"] == "message"
                message = opt["value"]
            end
        end
        
        respond!(bot, data; content = "$message, <@$user_id>!")
    end
end

start!(bot)

Command Types

Chat Input Commands (Slash Commands)

# Simple command
cmd = command("ping", "Check bot latency")

# Command with subcommands
cmd = command("settings", "Manage bot settings") do c
    subcommand(c, "view", "View current settings")
    subcommand(c, "reset", "Reset to defaults")
end

# Command with subcommand groups
cmd = command("admin", "Admin commands") do c
    subcommand_group(c, "user", "User management") do g
        subcommand(g, "ban", "Ban a user") do s
            option(s, "user", "User to ban";
                type = ApplicationCommandOptionTypes.USER,
                required = true
            )
        end
        subcommand(g, "kick", "Kick a user") do s
            option(s, "user", "User to kick";
                type = ApplicationCommandOptionTypes.USER,
                required = true
            )
        end
    end
end

Option Types

# String option
option(cmd, "name", "Your name";
    type = ApplicationCommandOptionTypes.STRING,
    required = true
)

# Integer option with min/max
option(cmd, "count", "Number of items";
    type = ApplicationCommandOptionTypes.INTEGER,
    required = true,
    min_value = 1,
    max_value = 100
)

# Choice option
option(cmd, "color", "Pick a color";
    type = ApplicationCommandOptionTypes.STRING,
    required = true,
    choices = [
        choice("Red", "red"),
        choice("Green", "green"),
        choice("Blue", "blue")
    ]
)

# User option
option(cmd, "target", "Select a user";
    type = ApplicationCommandOptionTypes.USER,
    required = true
)

# Channel option
option(cmd, "channel", "Select a channel";
    type = ApplicationCommandOptionTypes.CHANNEL,
    required = false
)

Deferred Responses

For commands that take time to process:

on!(bot, "INTERACTION_CREATE") do data
    if data["data"]["name"] == "slow-command"
        # Defer the response (shows "Bot is thinking...")
        defer!(bot, data)
        
        # Do slow work
        sleep(3)
        
        # Edit the deferred response
        edit_response!(bot, data; content = "Done processing!")
    end
end

Ephemeral Responses

Send responses only visible to the user:

on!(bot, "INTERACTION_CREATE") do data
    if data["data"]["name"] == "secret"
        respond!(bot, data;
            content = "This is only visible to you!",
            ephemeral = true
        )
    end
end

Follow-up Messages

Send additional messages after the initial response:

on!(bot, "INTERACTION_CREATE") do data
    if data["data"]["name"] == "multi"
        respond!(bot, data; content = "First message!")
        
        sleep(1)
        followup!(bot, data; content = "Second message!")
        
        sleep(1)
        followup!(bot, data; content = "Third message!")
    end
end

Guild-Specific Commands

Register commands for a specific guild (faster updates during development):

on!(bot, "READY") do data
    guild_id = Snowflake(123456789012345678)  # Your test guild ID
    
    cmd = command("test", "Test command")
    register_commands!(bot, [cmd]; guild_id = guild_id)
end

Context Menu Commands

# User context menu command
user_cmd = command("Get User Info";
    type = ApplicationCommandTypes.USER
)

# Message context menu command  
msg_cmd = command("Report Message";
    type = ApplicationCommandTypes.MESSAGE
)

register_commands!(bot, [user_cmd, msg_cmd])

on!(bot, "INTERACTION_CREATE") do data
    if data["type"] != InteractionTypes.APPLICATION_COMMAND
        return
    end
    
    cmd_type = data["data"]["type"]
    
    if cmd_type == ApplicationCommandTypes.USER
        # User context menu
        target_id = data["data"]["target_id"]
        respond!(bot, data; content = "User ID: $target_id")
        
    elseif cmd_type == ApplicationCommandTypes.MESSAGE
        # Message context menu
        target_id = data["data"]["target_id"]
        respond!(bot, data; content = "Message ID: $target_id")
    end
end
