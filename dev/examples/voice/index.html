Voice Example

This example demonstrates how to connect to voice channels and play audio.

Voice Dependencies

Voice support requires the optional Sodium.jl package for encryption:

using Pkg
Pkg.add("Sodium")

Basic Voice Connection

using Nobelium

bot = Bot(
    token = ENV["DISCORD_TOKEN"],
    intents = Intents.DEFAULT | Intents.GUILD_VOICE_STATES
)

# Join a voice channel
on!(bot, "MESSAGE_CREATE") do data
    if data["content"] == "!join"
        guild_id = Snowflake(data["guild_id"])
        
        # Get the user's voice state to find their channel
        # (You would typically get this from cache or API)
        channel_id = Snowflake(123456789012345678)  # Voice channel ID
        
        voice_conn = join_channel!(bot, guild_id, channel_id)
        reply(bot, data; content = "Joined voice channel!")
    end
end

# Leave voice channel
on!(bot, "MESSAGE_CREATE") do data
    if data["content"] == "!leave"
        guild_id = Snowflake(data["guild_id"])
        leave_channel!(bot, guild_id)
        reply(bot, data; content = "Left voice channel!")
    end
end

start!(bot)

Voice Connection Lifecycle

# The voice connection process:
# 1. Send Voice State Update to Gateway
# 2. Receive VOICE_STATE_UPDATE event
# 3. Receive VOICE_SERVER_UPDATE event
# 4. Connect to Voice WebSocket
# 5. Perform UDP discovery
# 6. Ready to send audio

on!(bot, "VOICE_STATE_UPDATE") do data
    println("Voice state updated for user: ", data["user_id"])
end

on!(bot, "VOICE_SERVER_UPDATE") do data
    println("Voice server ready: ", data["endpoint"])
end

Sending Audio

Audio must be in Opus format. Here's how to send pre-encoded Opus frames:

# Send a single Opus frame
function play_opus_frame(voice_conn::VoiceConnection, opus_data::Vector{UInt8})
    if is_ready(voice_conn)
        send_opus_frame(voice_conn, opus_data)
    end
end

# Play audio at the correct rate (20ms per frame)
function play_audio(voice_conn::VoiceConnection, frames::Vector{Vector{UInt8}})
    for frame in frames
        if !is_connected(voice_conn)
            break
        end
        
        send_opus_frame(voice_conn, frame)
        sleep(0.02)  # 20ms per frame
    end
end

Voice State Management

using Nobelium.Voice: VoiceStateManager, get_voice_state, is_in_voice

# Check if bot is in a voice channel
on!(bot, "MESSAGE_CREATE") do data
    if data["content"] == "!status"
        guild_id = Snowflake(data["guild_id"])
        
        if is_in_voice(guild_id)
            state = get_voice_state(guild_id)
            reply(bot, data; content = "In voice channel: $(state.channel_id)")
        else
            reply(bot, data; content = "Not in a voice channel")
        end
    end
end

Complete Voice Bot Example

using Nobelium

bot = Bot(
    token = ENV["DISCORD_TOKEN"],
    intents = Intents.DEFAULT | Intents.GUILD_VOICE_STATES | Intents.MESSAGE_CONTENT
)

# Store voice connections per guild
voice_connections = Dict{Snowflake, VoiceConnection}()

on!(bot, "MESSAGE_CREATE") do data
    content = get(data, "content", "")
    guild_id = haskey(data, "guild_id") ? Snowflake(data["guild_id"]) : nothing
    
    if guild_id === nothing
        return  # Ignore DMs
    end
    
    if startswith(content, "!join ")
        # Parse channel ID from command
        channel_id_str = content[7:end]
        channel_id = Snowflake(parse(UInt64, channel_id_str))
        
        try
            conn = join_channel!(bot, guild_id, channel_id)
            voice_connections[guild_id] = conn
            reply(bot, data; content = "Joined voice channel!")
        catch e
            reply(bot, data; content = "Failed to join: $(e)")
        end
        
    elseif content == "!leave"
        if haskey(voice_connections, guild_id)
            leave_channel!(bot, guild_id)
            delete!(voice_connections, guild_id)
            reply(bot, data; content = "Left voice channel!")
        else
            reply(bot, data; content = "Not in a voice channel!")
        end
        
    elseif content == "!voice-status"
        if haskey(voice_connections, guild_id)
            conn = voice_connections[guild_id]
            status = is_ready(conn) ? "Ready" : "Connecting..."
            reply(bot, data; content = "Voice status: $status")
        else
            reply(bot, data; content = "Not connected to voice")
        end
    end
end

start!(bot)

Encryption

Voice data is encrypted using XSalsa20-Poly1305:

using Nobelium.Voice: VoiceEncryption, encrypt_audio, decrypt_audio

# Encryption is handled automatically by the VoiceConnection
# The secret key is received during the voice handshake

# Manual encryption (if needed):
encryption = VoiceEncryption()
setup_encryption!(encryption, secret_key)

encrypted = encrypt_audio(encryption, audio_data, nonce)
decrypted = decrypt_audio(encryption, encrypted, nonce)

Tips

  1. Always check connection state before sending audio
  2. Use the correct frame timing (20ms per Opus frame)
  3. Handle disconnections gracefully - voice connections can drop
  4. Enable GUILDVOICESTATES intent to receive voice events
  5. Install Sodium.jl for voice encryption support
